import re
from dataclasses import dataclass
from typing import List

from app.llm import LLMService
from app.retrieval.citations import Citation, CitationBuilder
from app.retrieval.evidence import EvidenceAnalyzer, EvidenceResult
from app.retrieval.retriever import RetrievalService
from app.tools.order_lookup import OrderLookupService


# ==============================================================
# RESPONSE MODEL
# ==============================================================

@dataclass
class AgentResponse:
    answer_type: str
    message: str
    citations: List[Citation]
    evidence: EvidenceResult | None = None

    @property
    def is_grounded(self) -> bool:
        return self.answer_type == "grounded"

    @property
    def requires_handoff(self) -> bool:
        return self.answer_type == "conflict"


# ==============================================================
# SUPPORT AGENT
# ==============================================================

class SupportAgent:
    """
    Reliable Aster & Row customer support agent.

    Processing order:

    1. Detect explicit order IDs.
    2. Protect private/internal order information.
    3. Perform deterministic order lookup.
    4. Detect generic order questions.
    5. Retrieve knowledge-base evidence.
    6. Prioritize active/official evidence.
    7. Detect conflicts.
    8. Handle deterministic policy questions.
    9. Use LLM only when appropriate.
    10. Never allow the LLM to generate order information.

    Important safety properties:

    - Order information is NEVER generated by the LLM.
    - Internal/private order information is never exposed.
    - Time-specific return questions are not answered from
      unrelated evidence.
    - Shipping-address changes are NOT treated as private
      information requests.
    """

    ORDER_ID_PATTERN = re.compile(
        r"\bORD-\d+\b",
        re.IGNORECASE,
    )

    # ==========================================================
    # INITIALIZATION
    # ==========================================================

    def __init__(
        self,
        retrieval_service: RetrievalService,
        evidence_analyzer: EvidenceAnalyzer,
        llm_service: LLMService | None = None,
        order_lookup_service: OrderLookupService | None = None,
    ):
        self.retrieval_service = retrieval_service
        self.evidence_analyzer = evidence_analyzer

        self.llm_service = (
            llm_service
            or LLMService()
        )

        self.order_lookup_service = (
            order_lookup_service
            or OrderLookupService()
        )

    # ==========================================================
    # ORDER ID DETECTION
    # ==========================================================

    @classmethod
    def _extract_order_id(
        cls,
        query: str,
    ) -> str | None:

        match = cls.ORDER_ID_PATTERN.search(
            query
        )

        if not match:
            return None

        return match.group(0)

    @staticmethod
    def _is_order_question(
        query: str,
    ) -> bool:

        return (
            SupportAgent._extract_order_id(
                query
            )
            is not None
        )

    # ==========================================================
    # GENERIC ORDER QUESTION DETECTION
    # ==========================================================

    @staticmethod
    def _looks_like_order_question(
        query: str,
    ) -> bool:

        text = query.lower()

        order_terms = [
            "my order",
            "my shipment",
            "my package",
            "my delivery",
            "order status",
            "where is my order",
            "where's my order",
            "where is my package",
            "where's my package",
            "when will my order",
            "when will my package",
            "when will my shipment",
            "when does my order",
            "track my order",
            "track my package",
            "order arrive",
            "package arrive",
        ]

        return any(
            term in text
            for term in order_terms
        )

    # ============================================================
    # PRIVATE / INTERNAL INFORMATION DETECTION
    # ============================================================

    @staticmethod
    def _is_private_order_information_question(
        query: str,
    ) -> bool:
        """
        Detect requests for sensitive/private order information.

        Legitimate order-management questions such as:
            "Can I change my shipping address?"
            "Can I update my delivery address?"

        must NOT be blocked.
        """

        text = query.lower().strip()

        # --------------------------------------------------------
        # Internal risk information
        # --------------------------------------------------------

        risk_terms = [
            "risk score",
            "internal risk",
            "risk rating",
            "fraud score",
            "fraud risk",
            "internal score",
        ]

        if any(term in text for term in risk_terms):
            return True

        # --------------------------------------------------------
        # Internal/private notes
        # --------------------------------------------------------

        internal_terms = [
            "internal note",
            "internal notes",
            "internal comment",
            "internal comments",
            "staff note",
            "staff notes",
            "agent note",
            "agent notes",
            "internal information",
        ]

        if any(term in text for term in internal_terms):
            return True

        # --------------------------------------------------------
        # Customer email
        # --------------------------------------------------------

        email_terms = [
            "email address",
            "customer email",
            "email for",
            "email on the order",
        ]

        if any(term in text for term in email_terms):
            return True

        # --------------------------------------------------------
        # Shipping / delivery / billing address
        # --------------------------------------------------------

        address_terms = [
            "shipping address",
            "delivery address",
            "billing address",
        ]

        # These indicate that the customer wants to MODIFY
        # an address rather than reveal it.
        address_change_terms = [
            "change",
            "update",
            "modify",
            "correct",
            "edit",
        ]

        asks_for_address = any(
            term in text
            for term in address_terms
        )

        asks_to_change_address = any(
            term in text
            for term in address_change_terms
        )

        # Example:
        # "Can I change my shipping address?"
        # -> NOT private information
        if asks_for_address and not asks_to_change_address:

            address_request_terms = [
                "what is",
                "what's",
                "tell me",
                "show me",
                "give me",
                "provide",
                "where is",
                "where's",
            ]

            if any(
                term in text
                for term in address_request_terms
            ):
                return True

        return False
    # ==========================================================
    # PRIVATE INFORMATION RESPONSE
    # ==========================================================

    @staticmethod
    def _answer_private_information_request() -> AgentResponse:

        return AgentResponse(
            answer_type="insufficient_evidence",
            message=(
                "I can't provide internal or private customer "
                "information such as risk scores, email addresses, "
                "shipping addresses, or internal notes."
            ),
            citations=[],
            evidence=None,
        )

    # ==========================================================
    # ORDER LOOKUP
    # ==========================================================

    def _run_order_lookup(
        self,
        query: str,
    ) -> AgentResponse:

        order_id = self._extract_order_id(
            query
        )

        # ------------------------------------------------------
        # Missing order ID
        # ------------------------------------------------------

        if order_id is None:

            return AgentResponse(
                answer_type="clarification",
                message=(
                    "An order ID is required to look up "
                    "order information. Please provide "
                    "your order ID, for example ORD-1007."
                ),
                citations=[],
                evidence=None,
            )

        # ------------------------------------------------------
        # Deterministic order lookup
        # ------------------------------------------------------

        result = self.order_lookup_service.lookup(
            order_id
        )

        # ------------------------------------------------------
        # Unknown order
        # ------------------------------------------------------

        if not result.found:

            return AgentResponse(
                answer_type="order_not_found",
                message=(
                    result.message
                    or f"No order was found for {order_id}."
                ),
                citations=[],
                evidence=None,
            )

        # ------------------------------------------------------
        # Customer-safe response
        # ------------------------------------------------------

        parts = []

        status = (
            result.status or ""
        ).strip().lower()

        if status:

            parts.append(
                f"The status of {result.order_id} is "
                f"**{status}**."
            )

        # Customer-safe message supplied by lookup service.

        if result.customer_safe_message:

            parts.append(
                result.customer_safe_message
            )

        # ------------------------------------------------------
        # Carrier
        # ------------------------------------------------------

        if result.carrier:

            parts.append(
                f"Carrier: **{result.carrier}**."
            )

        # ------------------------------------------------------
        # Tracking number
        # ------------------------------------------------------

        if result.tracking_number:

            parts.append(
                "Tracking number: "
                f"**{result.tracking_number}**."
            )

        # ------------------------------------------------------
        # Estimated delivery
        # ------------------------------------------------------

        if result.estimated_delivery:

            parts.append(
                "Estimated delivery: "
                f"**{result.estimated_delivery}**."
            )

        message = " ".join(parts).strip()

        if not message:

            message = (
                f"Order {result.order_id} was found, "
                "but there is no customer-safe status "
                "information available."
            )

        return AgentResponse(
            answer_type="order_lookup",
            message=message,
            citations=[],
            evidence=None,
        )

    # ==========================================================
    # EVIDENCE PRIORITIZATION
    # ==========================================================

    @staticmethod
    def _is_active_official(
        result,
    ) -> bool:

        chunk = result.chunk

        status = (
            chunk.status or ""
        ).strip().lower()

        authority = (
            chunk.policy_authority or ""
        ).strip().lower()

        return (
            status == "active"
            and authority == "official"
        )

    @staticmethod
    def _is_active(
        result,
    ) -> bool:

        status = (
            result.chunk.status or ""
        ).strip().lower()

        return status == "active"

    def _select_best_evidence(
        self,
        evidence: EvidenceResult,
    ) -> EvidenceResult:

        chunks = evidence.chunks

        if not chunks:
            return evidence

        # ------------------------------------------------------
        # First preference:
        #
        # active + official
        # ------------------------------------------------------

        active_official = [
            result
            for result in chunks
            if self._is_active_official(result)
        ]

        if active_official:

            selected = active_official

        else:

            # --------------------------------------------------
            # Second preference:
            #
            # active sources
            # --------------------------------------------------

            active = [
                result
                for result in chunks
                if self._is_active(result)
            ]

            if active:
                selected = active
            else:
                selected = chunks

        # ------------------------------------------------------
        # Remove duplicate source/headings.
        #
        # Keep highest-scoring result.
        # ------------------------------------------------------

        unique = {}

        for result in selected:

            chunk = result.chunk

            key = (
                chunk.source_file,
                chunk.heading,
            )

            previous = unique.get(key)

            if (
                previous is None
                or result.score > previous.score
            ):
                unique[key] = result

        selected = list(
            unique.values()
        )

        # Highest relevance first.

        selected.sort(
            key=lambda item: item.score,
            reverse=True,
        )

        return EvidenceResult(
            chunks=selected,
            conflicts=evidence.conflicts,
            excluded_chunks=evidence.excluded_chunks,
        )

    # ==========================================================
    # POLICY QUESTION DETECTION
    # ==========================================================

    @staticmethod
    def _is_return_window_question(
        query: str,
    ) -> bool:

        text = query.lower()

        return (
            (
                "return" in text
                or "returns" in text
            )
            and (
                "days" in text
                or "window" in text
                or "how long" in text
            )
        )

    @staticmethod
    def _is_return_shipping_fee_question(
        query: str,
    ) -> bool:

        text = query.lower()

        return (
            (
                "return" in text
                or "returns" in text
            )
            and (
                "shipping fee" in text
                or "return fee" in text
                or "shipping cost" in text
            )
        )

    # ==========================================================
    # SHIPPING ADDRESS QUESTION
    # ==========================================================

    @staticmethod
    def _is_shipping_address_question(
        query: str,
    ) -> bool:

        text = query.lower()

        return (
            "shipping address" in text
            or "delivery address" in text
            or "change my address" in text
            or "change the address" in text
            or "update my address" in text
            or "update the address" in text
            or "modify my address" in text
            or "edit my address" in text
        )

    # ==========================================================
    # TIME-SPECIFIC RETURN QUESTION
    # ==========================================================

    @staticmethod
    def _is_time_specific_return_question(
        query: str,
    ) -> bool:

        text = query.lower()

        return (
            (
                "return" in text
                or "returns" in text
            )
            and bool(
                re.search(
                    r"\b\d+\s*days?\b",
                    text,
                )
            )
        )

    # ==========================================================
    # FINAL-SALE QUESTION
    # ==========================================================

    @staticmethod
    def _is_final_sale_question(
        query: str,
    ) -> bool:

        text = query.lower()

        return (
            (
                "final-sale" in text
                or "final sale" in text
            )
            and (
                "return" in text
                or "exchange" in text
            )
        )

    # ==========================================================
    # DETERMINISTIC RETURN WINDOW ANSWER
    # ==========================================================

    def _answer_return_window(
        self,
        evidence: EvidenceResult,
    ) -> AgentResponse | None:

        trailplus_chunk = None
        standard_chunk = None

        for result in evidence.chunks:

            text = (
                result.chunk.content or ""
            ).lower()

            if (
                "trailplus" in text
                and (
                    "45" in text
                    or "forty-five" in text
                )
            ):
                trailplus_chunk = result

            if (
                "standard" in text
                and (
                    "30" in text
                    or "thirty" in text
                )
            ):
                standard_chunk = result

        # ------------------------------------------------------
        # Both policies
        # ------------------------------------------------------

        if (
            trailplus_chunk
            and standard_chunk
        ):

            citations = CitationBuilder.build_many(
                [
                    trailplus_chunk,
                    standard_chunk,
                ]
            )

            message = (
                "The return window depends on your "
                "membership status:\n\n"
                "- **TrailPlus members:** 45 calendar days "
                "from delivery for eligible items, provided "
                "the membership was active when the order "
                "was placed.\n"
                "- **Standard plan customers:** 30 calendar "
                "days from delivery.\n\n"
                "Joining TrailPlus after placing an order "
                "does not extend that order's return window."
            )

            return AgentResponse(
                answer_type="grounded",
                message=message,
                citations=citations,
                evidence=evidence,
            )

        # ------------------------------------------------------
        # TrailPlus only
        # ------------------------------------------------------

        if trailplus_chunk:

            citations = CitationBuilder.build_many(
                [trailplus_chunk]
            )

            message = (
                "TrailPlus members receive a "
                "**45-calendar-day return window from "
                "delivery** for eligible items, provided "
                "the membership was active when the order "
                "was placed.\n\n"
                "Joining TrailPlus after placing an order "
                "does not extend that order's return window."
            )

            return AgentResponse(
                answer_type="grounded",
                message=message,
                citations=citations,
                evidence=evidence,
            )

        # ------------------------------------------------------
        # Standard only
        # ------------------------------------------------------

        if standard_chunk:

            citations = CitationBuilder.build_many(
                [standard_chunk]
            )

            message = (
                "Standard plan customers have a "
                "**30-calendar-day return window from "
                "delivery** for eligible items."
            )

            return AgentResponse(
                answer_type="grounded",
                message=message,
                citations=citations,
                evidence=evidence,
            )

        return None

    # ==========================================================
    # DETERMINISTIC RETURN SHIPPING ANSWER
    # ==========================================================

    def _answer_return_shipping_fee(
        self,
        evidence: EvidenceResult,
    ) -> AgentResponse | None:

        policy_chunk = None

        for result in evidence.chunks:

            text = (
                result.chunk.content or ""
            ).lower()

            if (
                "$6.95" in text
                or "6.95" in text
            ):
                policy_chunk = result
                break

        if policy_chunk is None:
            return None

        citations = CitationBuilder.build_many(
            [policy_chunk]
        )

        message = (
            "The return shipping fee for standard domestic "
            "returns is **$6.95**. The fee is deducted from "
            "the refund unless the item arrived damaged or "
            "was sent incorrectly, in which case the fee "
            "is waived."
        )

        return AgentResponse(
            answer_type="grounded",
            message=message,
            citations=citations,
            evidence=evidence,
        )

    # ==========================================================
    # DETERMINISTIC FINAL-SALE ANSWER
    # ==========================================================

    def _answer_final_sale(
        self,
        evidence: EvidenceResult,
    ) -> AgentResponse | None:

        relevant = []

        for result in evidence.chunks:

            text = (
                result.chunk.content or ""
            ).lower()

            if (
                (
                    "final sale" in text
                    or "final-sale" in text
                )
                and (
                    "return" in text
                    or "exchange" in text
                )
            ):
                relevant.append(result)

        if not relevant:
            return None

        # Highest-scoring evidence first.

        relevant.sort(
            key=lambda item: item.score,
            reverse=True,
        )

        # Keep only strongest two citations.

        selected = relevant[:2]

        citations = CitationBuilder.build_many(
            selected
        )

        message = (
            "Final-sale items cannot be returned or "
            "exchanged for a change of mind. However, "
            "damaged, defective, or incorrect final-sale "
            "items may still qualify for assistance under "
            "the Damaged or Wrong Items Policy."
        )

        return AgentResponse(
            answer_type="grounded",
            message=message,
            citations=citations,
            evidence=evidence,
        )

    # ==========================================================
    # DETERMINISTIC SHIPPING ADDRESS ANSWER
    # ==========================================================

    def _answer_shipping_address(
        self,
        evidence: EvidenceResult,
    ) -> AgentResponse | None:

        address_chunk = None

        for result in evidence.chunks:

            chunk = result.chunk

            source = (
                chunk.source_file or ""
            ).lower()

            heading = (
                chunk.heading or ""
            ).lower()

            content = (
                chunk.content or ""
            ).lower()

            # --------------------------------------------------
            # Exact address-change policy is preferred.
            # --------------------------------------------------

            if (
                "order-changes-and-cancellations.md"
                in source
                and "address" in heading
            ):
                address_chunk = result
                break

            if (
                "address changes" in heading
                or "shipping address" in content
            ):

                if address_chunk is None:
                    address_chunk = result

        if address_chunk is None:
            return None

        citations = CitationBuilder.build_many(
            [address_chunk]
        )

        message = (
            "You can request a change to your shipping "
            "address within **30 minutes** while an order "
            "is still pending. Once the order enters "
            "processing, Aster & Row cannot guarantee the "
            "change. After shipment, address modifications "
            "must be handled through the carrier."
        )

        return AgentResponse(
            answer_type="grounded",
            message=message,
            citations=citations,
            evidence=evidence,
        )

    # ==========================================================
    # GENERIC GROUNDED RESPONSE
    # ==========================================================

    def _generic_grounded_response(
        self,
        evidence: EvidenceResult,
    ) -> AgentResponse:

        citations = CitationBuilder.build_many(
            evidence.chunks
        )

        return AgentResponse(
            answer_type="grounded",
            message="Relevant evidence was found.",
            citations=citations,
            evidence=evidence,
        )

    # ==========================================================
    # RETRIEVE + ANALYZE EVIDENCE
    # ==========================================================

    def _retrieve_evidence(
        self,
        query: str,
        top_k: int,
    ) -> EvidenceResult:

        retrieved = (
            self.retrieval_service.retrieve(
                query,
                top_k=top_k,
            )
        )

        evidence = (
            self.evidence_analyzer.analyze(
                retrieved
            )
        )

        evidence = self._select_best_evidence(
            evidence
        )

        return evidence

    # ==========================================================
    # DETERMINISTIC ANSWER ROUTER
    # ==========================================================

    def _try_deterministic_answer(
        self,
        query: str,
        evidence: EvidenceResult,
    ) -> AgentResponse | None:

        # ------------------------------------------------------
        # Final-sale
        # ------------------------------------------------------

        if self._is_final_sale_question(query):

            answer = self._answer_final_sale(
                evidence
            )

            if answer is not None:
                return answer

        # ------------------------------------------------------
        # Shipping address
        # ------------------------------------------------------

        if self._is_shipping_address_question(query):

            answer = self._answer_shipping_address(
                evidence
            )

            if answer is not None:
                return answer

        # ------------------------------------------------------
        # Time-specific return question
        #
        # IMPORTANT:
        #
        # "Can I return an item after 50 days?"
        #
        # must NOT be answered using unrelated final-sale
        # evidence.
        # ------------------------------------------------------

        if self._is_time_specific_return_question(
            query
        ):

            # Try return-window evidence.

            if self._is_return_window_question(
                query
            ):

                answer = self._answer_return_window(
                    evidence
                )

                # --------------------------------------------------
                # Check requested number of days.
                #
                # Example:
                #
                # supported = 30, 45
                # requested = 50
                #
                # Therefore:
                #
                # insufficient_evidence
                # --------------------------------------------------

                if answer is not None:

                    requested_days = re.search(
                        r"\b(\d+)\s*days?\b",
                        query.lower(),
                    )

                    if requested_days:

                        requested = int(
                            requested_days.group(1)
                        )

                        supported = []

                        for result in evidence.chunks:

                            content = (
                                result.chunk.content
                                or ""
                            ).lower()

                            if (
                                "45" in content
                                and "trailplus" in content
                            ):
                                supported.append(45)

                            if (
                                "30" in content
                                and "standard" in content
                            ):
                                supported.append(30)

                        # --------------------------------------------------
                        # Critical safety rule:
                        #
                        # Never answer "50 days" with
                        # a 30/45-day policy.
                        # --------------------------------------------------

                        if (
                            supported
                            and requested not in supported
                        ):

                            return AgentResponse(
                                answer_type=(
                                    "insufficient_evidence"
                                ),
                                message=(
                                    "I don't have enough "
                                    "information in the "
                                    "knowledge base to answer "
                                    "this question reliably."
                                ),
                                citations=[],
                                evidence=None,
                            )

                    return answer

            # No supported return-window evidence.

            return AgentResponse(
                answer_type="insufficient_evidence",
                message=(
                    "I don't have enough information in the "
                    "knowledge base to answer this question "
                    "reliably."
                ),
                citations=[],
                evidence=None,
            )

        # ------------------------------------------------------
        # Normal return-window question
        # ------------------------------------------------------

        if self._is_return_window_question(query):

            answer = self._answer_return_window(
                evidence
            )

            if answer is not None:
                return answer

        # ------------------------------------------------------
        # Return shipping fee
        # ------------------------------------------------------

        if self._is_return_shipping_fee_question(
            query
        ):

            answer = self._answer_return_shipping_fee(
                evidence
            )

            if answer is not None:
                return answer

        return None

    # ==========================================================
    # MAIN RAG PIPELINE
    # ==========================================================

    def run(
        self,
        query: str,
        top_k: int = 5,
    ) -> AgentResponse:

        query = query.strip()

        if not query:
            raise ValueError(
                "Query cannot be empty."
            )

        # ------------------------------------------------------
        # Explicit order ID
        # ------------------------------------------------------

        if self._is_order_question(query):

            # --------------------------------------------------
            # NEVER expose private/internal information.
            # --------------------------------------------------

            if self._is_private_order_information_question(
                query
            ):

                return (
                    self._answer_private_information_request()
                )

            # --------------------------------------------------
            # Safe order lookup.
            # --------------------------------------------------

            return self._run_order_lookup(
                query
            )

        # ------------------------------------------------------
        # Generic order question
        # ------------------------------------------------------

        if self._looks_like_order_question(query):

            return self._run_order_lookup(
                query
            )

        # ------------------------------------------------------
        # Protected information without order ID
        # ------------------------------------------------------

        if self._is_private_order_information_question(
            query
        ):

            return (
                self._answer_private_information_request()
            )

        # ------------------------------------------------------
        # Retrieve evidence
        # ------------------------------------------------------

        evidence = self._retrieve_evidence(
            query,
            top_k,
        )

        # ------------------------------------------------------
        # Conflict
        # ------------------------------------------------------

        if evidence.has_conflict:

            citations = CitationBuilder.build_many(
                evidence.chunks
            )

            return AgentResponse(
                answer_type="conflict",
                message=(
                    "I found conflicting current official "
                    "information. I cannot safely provide "
                    "a definitive answer without resolving "
                    "the conflict."
                ),
                citations=citations,
                evidence=evidence,
            )

        # ------------------------------------------------------
        # No evidence
        # ------------------------------------------------------

        if not evidence.chunks:

            return AgentResponse(
                answer_type="insufficient_evidence",
                message=(
                    "I don't have enough information "
                    "in the knowledge base to answer "
                    "this question reliably."
                ),
                citations=[],
                evidence=evidence,
            )

        # ------------------------------------------------------
        # Deterministic answers
        # ------------------------------------------------------

        deterministic = (
            self._try_deterministic_answer(
                query,
                evidence,
            )
        )

        if deterministic is not None:
            return deterministic

        # ------------------------------------------------------
        # Generic grounded response
        # ------------------------------------------------------

        return self._generic_grounded_response(
            evidence
        )

    # ==========================================================
    # LLM PIPELINE
    # ==========================================================

    def run_with_llm(
        self,
        query: str,
        top_k: int = 5,
    ) -> AgentResponse:

        query = query.strip()

        if not query:
            raise ValueError(
                "Query cannot be empty."
            )

        # ------------------------------------------------------
        # Explicit order ID
        #
        # NEVER send order questions to the LLM.
        # ------------------------------------------------------

        if self._is_order_question(query):

            if self._is_private_order_information_question(
                query
            ):

                return (
                    self._answer_private_information_request()
                )

            return self._run_order_lookup(
                query
            )

        # ------------------------------------------------------
        # Generic order question
        # ------------------------------------------------------

        if self._looks_like_order_question(query):

            return self._run_order_lookup(
                query
            )

        # ------------------------------------------------------
        # Protected information
        # ------------------------------------------------------

        if self._is_private_order_information_question(
            query
        ):

            return (
                self._answer_private_information_request()
            )

        # ------------------------------------------------------
        # Retrieve evidence
        # ------------------------------------------------------

        evidence = self._retrieve_evidence(
            query,
            top_k,
        )

        # ------------------------------------------------------
        # Conflict
        # ------------------------------------------------------

        if evidence.has_conflict:

            citations = CitationBuilder.build_many(
                evidence.chunks
            )

            return AgentResponse(
                answer_type="conflict",
                message=(
                    "I found conflicting current official "
                    "information. I cannot safely provide "
                    "a definitive answer without resolving "
                    "the conflict."
                ),
                citations=citations,
                evidence=evidence,
            )

        # ------------------------------------------------------
        # Insufficient evidence
        # ------------------------------------------------------

        if not evidence.chunks:

            return AgentResponse(
                answer_type="insufficient_evidence",
                message=(
                    "I don't have enough information "
                    "in the knowledge base to answer "
                    "this question reliably."
                ),
                citations=[],
                evidence=evidence,
            )

        # ------------------------------------------------------
        # Deterministic policy answers
        # ------------------------------------------------------

        deterministic = (
            self._try_deterministic_answer(
                query,
                evidence,
            )
        )

        if deterministic is not None:
            return deterministic

        # ------------------------------------------------------
        # Generic LLM response
        #
        # The LLM receives ONLY retrieved evidence.
        # It never receives the complete knowledge base
        # or orders dataset.
        # ------------------------------------------------------

        citations = CitationBuilder.build_many(
            evidence.chunks
        )

        try:

            llm_message = (
                self.llm_service.generate_answer(
                    query=query,
                    evidence=evidence,
                )
            )

        except TypeError:

            # Compatibility fallback for implementations
            # using positional parameters.

            try:

                llm_message = (
                    self.llm_service.generate_answer(
                        query,
                        evidence,
                    )
                )

            except Exception:

                llm_message = None

        except Exception:

            llm_message = None

        # ------------------------------------------------------
        # LLM failure must NEVER produce hallucination.
        # ------------------------------------------------------

        if not llm_message:

            return AgentResponse(
                answer_type="grounded",
                message="Relevant evidence was found.",
                citations=citations,
                evidence=evidence,
            )

        return AgentResponse(
            answer_type="grounded",
            message=str(
                llm_message
            ).strip(),
            citations=citations,
            evidence=evidence,
        )
